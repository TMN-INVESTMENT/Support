// Format mobile number
function formatMobile($num) {
    $num = preg_replace('/[^0-9]/', '', $num);
    if (substr($num, 0, 1) === '0') {
        return '255' . substr($num, 1);
    } elseif (substr($num, 0, 1) === '7') {
        return '255' . $num;
    }
    return $num;
}
